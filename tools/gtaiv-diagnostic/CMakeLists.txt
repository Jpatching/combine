cmake_minimum_required(VERSION 3.20)
project(combine_iv_diagnostic LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "This diagnostic requires 32-bit Windows")
endif()
add_library(combine_iv_diagnostic SHARED diagnostic.cpp)
target_compile_features(combine_iv_diagnostic PRIVATE cxx_std_17)
target_link_libraries(combine_iv_diagnostic PRIVATE version)
set_target_properties(combine_iv_diagnostic PROPERTIES PREFIX "" SUFFIX ".cleo")
if(MINGW)
  target_compile_options(combine_iv_diagnostic PRIVATE -Wall -Wextra -Werror)
  target_link_options(combine_iv_diagnostic PRIVATE -static -static-libgcc -static-libstdc++)
endif()
